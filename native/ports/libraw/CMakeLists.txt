cmake_minimum_required(VERSION 3.22)
project(phaos_libraw VERSION 0.22.2 LANGUAGES CXX)
include(GNUInstallDirs)
find_package(JPEG REQUIRED)
find_package(ZLIB REQUIRED)
file(GLOB_RECURSE raw_sources CONFIGURE_DEPENDS src/*.cpp)
list(FILTER raw_sources EXCLUDE REGEX "_ph\\.cpp$")
add_library(raw_r STATIC ${raw_sources})
set_target_properties(raw_r PROPERTIES CXX_STANDARD 11 CXX_STANDARD_REQUIRED ON CXX_EXTENSIONS OFF)
target_compile_definitions(raw_r PRIVATE USE_JPEG USE_ZLIB)
target_include_directories(raw_r PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>)
target_link_libraries(raw_r PRIVATE JPEG::JPEG ZLIB::ZLIB)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(raw_r PRIVATE -O3)
endif()
# Decoder sources stay serial/reentrant; the Cargo shim has its own bounded team.
install(TARGETS raw_r ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY libraw DESTINATION ${CMAKE_INSTALL_INCLUDEDIR} FILES_MATCHING PATTERN "*.h")
configure_file(${CMAKE_CURRENT_SOURCE_DIR}/phaos-libraw.pc.in phaos-libraw.pc @ONLY)
install(FILES ${CMAKE_CURRENT_BINARY_DIR}/phaos-libraw.pc DESTINATION ${CMAKE_INSTALL_LIBDIR}/pkgconfig)
